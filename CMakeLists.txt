cmake_minimum_required(VERSION 3.22)
set(LUMEN_VERSION "1.4.0" CACHE STRING "Release version")
set(LUMEN_REPOSITORY "AuronNetwork/Lumen" CACHE STRING "GitHub update repository")
project(Lumen VERSION ${LUMEN_VERSION} LANGUAGES C CXX RC)
configure_file(src/version.h.in generated/version.h @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(Lumen SHARED src/client.cpp src/overlay.cpp src/version.rc)
target_link_libraries(Lumen PRIVATE minhook user32 gdi32 comctl32 version d3d11 d3d12 dxgi d2d1 dwrite)
target_compile_definitions(Lumen PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX LUMEN_CLIENT_DLL)
target_compile_options(Lumen PRIVATE /W4 /EHsc /utf-8 /O2)
target_link_options(Lumen PRIVATE /DEBUG /OPT:REF /OPT:ICF)
add_executable(LumenLauncher WIN32 src/launcher.cpp src/launcher_ui.cpp src/updater.cpp src/version.rc)
set_target_properties(LumenLauncher PROPERTIES OUTPUT_NAME Lumen)
target_compile_definitions(LumenLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LumenLauncher PRIVATE /W4 /EHsc /utf-8 /O2)
target_link_libraries(LumenLauncher PRIVATE user32 shell32 advapi32 comctl32 dwmapi d2d1 dwrite winhttp bcrypt)
add_executable(scanner_tests tests/scanner_tests.cpp)
target_compile_options(scanner_tests PRIVATE /W4 /EHsc /utf-8)
enable_testing()
add_test(NAME scanner COMMAND scanner_tests)

add_executable(view_tests tests/view_tests.cpp)
target_compile_options(view_tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME view_features COMMAND view_tests)

add_executable(mouse_tests tests/mouse_tests.cpp)
target_compile_options(mouse_tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME mouse_input COMMAND mouse_tests)

# Isolated visual harness: never linked to the injector and not shipped as a binary.
add_executable(launcher_preview EXCLUDE_FROM_ALL WIN32 tests/launcher_preview.cpp src/launcher_ui.cpp)
target_compile_definitions(launcher_preview PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(launcher_preview PRIVATE /W4 /EHsc /utf-8 /O2)
target_link_libraries(launcher_preview PRIVATE user32 shell32 comctl32 dwmapi d2d1 dwrite)

add_executable(update_tests tests/update_tests.cpp)
target_compile_options(update_tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME update_policy COMMAND update_tests)

add_executable(updater_io_tests tests/updater_io_tests.cpp src/updater.cpp)
target_compile_definitions(updater_io_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(updater_io_tests PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(updater_io_tests PRIVATE winhttp bcrypt)
add_test(NAME updater_hash COMMAND updater_io_tests)
